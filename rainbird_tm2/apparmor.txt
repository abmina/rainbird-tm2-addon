#include <tunables/global>

# AppArmor profile for the Rain Bird ESP-TM2 add-on.
# The Supervisor renames the profile to the add-on's full slug when loading it.
profile rainbird_tm2 flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities needed by the s6-overlay init of the Home Assistant base image
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,

  # Networking: HTTPS to the Rain Bird module, MQTT, Supervisor API, web UI, UDP discovery
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,

  signal (send,receive) set=(kill,term,int,hup,cont,usr1,usr2),

  # General file access (as in Home Assistant's add-on profile template); program
  # execution below is still limited to the listed paths.
  file,

  # s6-overlay (base image init)
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/lib/** mr,
  /lib/** mr,
  /package/** ix,
  /command/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /etc/s6-overlay/** rwix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/null rw,
  /dev/urandom r,
  /tmp/{,**} rwk,

  # Read-only system configuration (time zones, TLS roots, DNS, ICU data)
  /etc/** r,
  /usr/share/** r,
  /proc/** r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/cpu/** r,

  # The add-on itself: program files read-only, its own data folder read-write
  /run.sh rix,
  /app/** r,
  /data/{,**} rw,
}
